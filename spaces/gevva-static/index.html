⚡Quickstart in Python

🔮Visual Claim Verification 

from PIL import Image
from gevva import GevvaCrossEncoder

# 1. Load multimodal model
model = GevvaCrossEncoder("davidburhans/gevva-e2b-multimodal")

# 2. Check claim against invoice image in 16.5 ms
image = Image.open("invoice.jpg").convert("RGB")
claim = "The total balance due is exactly $1,420.50."

pred = model.predict(
    pairs=[("An invoice is displayed.", claim)],
    images=[image]
)[0]

print(pred.predicted_label)  # 'entailment'
print(pred.probabilities)    # [0.03, 0.94, 0.03]

🎯Zero-Shot Tool & Intent Routing 

from gevva import GevvaCrossEncoder

model = GevvaCrossEncoder("davidburhans/gevva-e2b")

# Instant routing without prompt latency
tools = [
    "process_refund: Refund payment to customer card",
    "track_order: Query courier package tracking",
    "cancel_order: Cancel pending warehouse shipment",
]

result = model.route(
    query="I never got my package #9924, please refund me",
    tools=tools
)
print("Route to:", result.selected_tool)
# Output: 'process_refund'
